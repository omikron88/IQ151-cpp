cmake_minimum_required(VERSION 3.16)
project(iq151 VERSION 1.0.0 LANGUAGES C CXX)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(IQ151_BUILD_GUI "Build SDL2 desktop interface" ON)
option(IQ151_STATIC "Link SDL and C++ runtime statically (only OS libraries remain external)" OFF)
option(IQ151_FETCH_SDL "Build a pinned SDL2 if a suitable system package is unavailable" ON)
include(CTest)
include(cmake/EmbedAssets.cmake)
add_library(iq151_core src/machine.cpp src/media.cpp src/files.cpp src/sdrom.cpp src/snapshot.cpp src/disassembler.cpp "${ASSET_CPP}")
target_include_directories(iq151_core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(iq151_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(iq151 src/main.cpp)
target_link_libraries(iq151 PRIVATE iq151_core)
if(IQ151_BUILD_GUI)
    if(NOT IQ151_STATIC)
        find_package(SDL2 2.0.18 QUIET CONFIG)
    endif()
    if(IQ151_STATIC OR NOT TARGET SDL2::SDL2)
        if(NOT IQ151_FETCH_SDL)
            message(FATAL_ERROR "SDL2 >= 2.0.18 is required. Install libsdl2-dev or enable IQ151_FETCH_SDL.")
        endif()
        include(FetchContent)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        set(SDL_VULKAN OFF CACHE BOOL "Vulkan is not needed by the SDL renderer frontend")
        FetchContent_Declare(SDL2
            URL https://codeload.github.com/libsdl-org/SDL/tar.gz/5d249570393f7a37e037abf22cd6012a4cc56a71
            URL_HASH SHA256=10f1194f8d2e4a73ca1c7c553b3189d0b68ab9ef3544e8d2a268e4429456b373
            TLS_VERIFY ON) # SDL 2.32.10
        FetchContent_MakeAvailable(SDL2)
        set(IQ151_SDL SDL2::SDL2-static)
    else()
        set(IQ151_SDL SDL2::SDL2)
    endif()
    add_library(iq151_imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
        third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
        third_party/imgui/backends/imgui_impl_sdl2.cpp third_party/imgui/backends/imgui_impl_sdlrenderer2.cpp)
    target_include_directories(iq151_imgui PUBLIC third_party/imgui third_party/imgui/backends)
    target_link_libraries(iq151_imgui PUBLIC ${IQ151_SDL})
    target_sources(iq151 PRIVATE src/frontend.cpp src/ui_window.cpp)
    target_link_libraries(iq151 PRIVATE iq151_imgui)
    target_compile_definitions(iq151 PRIVATE IQ151_GUI SDL_MAIN_HANDLED)
endif()
if(IQ151_STATIC AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    if(WIN32)
        target_link_options(iq151 PRIVATE -static -static-libgcc -static-libstdc++)
    else()
        target_link_options(iq151 PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(iq151_tests tests/core_tests.cpp)
    target_link_libraries(iq151_tests PRIVATE iq151_core)
    target_compile_definitions(iq151_tests PRIVATE IQ151_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME core COMMAND iq151_tests)
    add_test(NAME boot COMMAND iq151 --headless --frames 120 --expect "BASIC")
    if(IQ151_BUILD_GUI)
        add_test(NAME sdl_smoke COMMAND iq151 --ui-smoke)
        add_test(NAME sdl_windows COMMAND iq151 --window-smoke)
        set_tests_properties(sdl_smoke sdl_windows PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    endif()
endif()
install(TARGETS iq151 RUNTIME DESTINATION bin)
install(FILES LICENSE NOTICE README.md DESTINATION share/doc/iq151)
install(FILES third_party/imgui/LICENSE.txt DESTINATION share/doc/iq151 RENAME ImGui-LICENSE.txt)
install(FILES third_party/SDL2-LICENSE.txt third_party/GCC-RUNTIME-LIBRARY-EXCEPTION.txt DESTINATION share/doc/iq151)
set(CPACK_PACKAGE_NAME iq151)
set(CPACK_GENERATOR ZIP TGZ)
include(CPack)
